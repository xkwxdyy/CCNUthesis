\chapter{高中数学实验教学资源课程讲义示例}
\section{讲义1——平面向量基本定理实验讲义}
\begin{description}
  \item[题目] 
    \hspace*{2em} 平面向量基本定理交互演示
  \item[目的] \mbox{} \vspace*{-1.8em}
    \begin{enumerate}
      \item 知识目标
        \begin{enumerate}
          \item 理解平面向量基本定理的内容：两条不共线的向量可以线性表示平面内的任意向量。
          \item 掌握用两条不共线的向量表示平面内任意向量的方法。
        \end{enumerate}
      \item 能力目标
        \begin{enumerate}
          \item 培养空间想象能力、逻辑推理能力以及动手操作能力。
          \item 提升运用数学工具（如GeoGebra）解决数学问题的能力。
        \end{enumerate}
      \item 情感目标
        \begin{enumerate}
          \item 激发对数学实验的兴趣，增强数学学习的参与感和成就感。
          \item 培养合作意识和探究精神。
        \end{enumerate}
    \end{enumerate}
  \item[器材] \hspace*{2em} 电脑、GeoGebra软件
  \item[过程] \mbox{} \vspace{-1.8em}
    \begin{enumerate}[label=(\zhnum*)]
      \item 主要流程（预演）
        \begin{enumerate}[1.]
          \item 引入新课
          \begin{itemize}
              \item 教师活动：配合 PPT 课件，引入课题，步步探究，引发思考—如何用两条不共线的向量表示平面内的任意向量。
              \item 学生活动：跟随老师讲课思路，步步探究，发散思考。
          \end{itemize}
          \item 实验演示
          \begin{itemize}
              \item 教师活动：GeoGebra 实验演示平面向量基本定理，让学生直观理解。
              \item 学生活动：通过实验理解平面向量基本定理，交互感知并消化吸收。
          \end{itemize}
          \item 定理阐述
          \begin{itemize}
              \item 教师活动：给出平面向量基本定理的文字表述，重点解读。
              \item 学生活动：理解并加以记忆。
          \end{itemize}
          \item 知识小结
          \begin{itemize}
              \item 教师活动：回顾新知，总结升华。
              \item 学生活动：回顾课堂，温故知新。
          \end{itemize}
          \item 课堂练习
          \begin{itemize}
              \item 教师活动：布置练习题，要求运用平面向量基本定理解决问题。
              \item 学生活动：完成课堂练习题，巩固对平面向量基本定理的理解。
          \end{itemize}
        \end{enumerate}
      \item 实验演示
      
        \hspace*{2em}教师活动：
        \begin{enumerate}
            \item 打开 GeoGebra，点击操作步骤按钮，展示两条不共线的向量\(\overrightarrow{OA}\)和\(\overrightarrow{OB}\)、平面内的任意向量\(\overrightarrow{a}\)，以及分解过程，见图 \ref{fig:教师演示界面示意图}。
            \begin{figure}[htbp]
              \centering
              \includegraphics[width=0.8\textwidth]{教师演示界面示意图}
              \caption{教师演示界面示意图}
              \label{fig:教师演示界面示意图}
            \end{figure}
            
            \item 通过调整向量\(\overrightarrow{a}\)的终点\(\mathrm{C}\)的位置，动态展示如何用\(\overrightarrow{OA}\)和\(\overrightarrow{OB}\)的线性组合表示\(\overrightarrow{a}\)。
            \item 强调向量线性组合的几何意义：\(\overrightarrow{a}=\lambda\overrightarrow{OM}+\mu\overrightarrow{ON}\)，其中\(\lambda\)和\(\mu\)是实数。
            \item 拖动合成向量的终点\(\mathrm{C}\)，演示不同位置的向量\(\overrightarrow{a}\)，展示\(\lambda\)和\(\mu\)的变化。
        \end{enumerate}

        \hspace*{2em}学生活动：
        \begin{enumerate}
            \item 观察教师的演示，理解平面向量基本定理的几何意义。
            \item 思考如何通过调整\(\overrightarrow{OA}\)和\(\overrightarrow{OB}\)表示不同的向量\(\overrightarrow{a}\)。
        \end{enumerate}
    \end{enumerate}
  \item[结论] \hspace*{2em} 相关实验教学期待学生能够理解平面向量基本定理的几何意义, 掌握用两条不共线的向量表示平面内任意向量的方法。通过实验操作, 直观可视化地理解平面向量基本定理, 培养学生的空间想象能力和逻辑推理能力。

  在实际教学中, 通过 GeoGebra 的动态演示和学生的动手操作, 学生能够直观理解平面向量基本定理, 教学目标基本达成。
  \item[分析]
  \hspace*{2em} 可在课本基础练习题的基础上加以启发式问题, 如下:
  \begin{enumerate}
    \item 如果两条向量共线, 是否还能表示平面内的任意向量? 为什么?
    \item 在三维空间中, 是否可以用三条不共面的向量表示空间内的任意向量? 尝试用 GeoGebra 进行验证。
  \end{enumerate}
\end{description}


\section{讲义2——三角函数图象平移伸缩变换实验讲义}
\begin{description}
  \item[题目]\hspace*{2em} 三角函数图象平移伸缩变换
  \item[目的] \mbox{} \vspace*{-1.8em}
    \begin{enumerate}
      \item 知识目标
        \begin{enumerate}
          \item 理解相位平移、周期伸缩、振幅变换对三角函数图象的影响
          \item 掌握函数 \(y = A\sin \left( {\omega x + \varphi }\right)\) 中各参数对应的图象变换规律
        \end{enumerate}
      \item 能力目标
        \begin{enumerate}
          \item 培养函数图象变换的直观感知能力与代数表达式对应关系
          \item 提升用 GeoGebra 进行数学实验和动态验证的能力
        \end{enumerate}
      \item 情感目标
        \begin{enumerate}
          \item 通过动态演示感受数学变换之美，激发学习兴趣
          \item 培养数形结合的思想方法和严谨的数学思维
        \end{enumerate}
    \end{enumerate}
  \item[器材] \hspace*{2em} 电脑、GeoGebra软件
  \item[过程] \mbox{} \vspace{-1.8em}
    \begin{enumerate}[label=(\zhnum*)]
      \item 主要流程（预演）
        \begin{enumerate}[1.]
          \item 引入新课
          \begin{itemize}
              \item 教师活动：可通过展示正弦曲线在物理振动、声波等领域的应用实例，引出参数变化对波形的影响。
              \item 学生活动：观察典型波形图，思考参数变化与图象特征的对应关系。
          \end{itemize}
          \item 实验演示
          \begin{itemize}
              \item 教师活动：分模块演示四种变换类型，引导学生观察参数与图象的对应关系。
              \item 学生活动：记录不同参数取值时的图象特征，总结变换规律。
          \end{itemize}
          \item 定理阐述
          \begin{itemize}
              \item 教师活动：系统讲解相位平移量、周期计算公式、振幅定义等数学原理。
              \item 学生活动：建立参数变化与函数表达式之间的代数对应关系。
          \end{itemize}
          \item 知识小结
          \begin{itemize}
              \item 教师活动：用坐标系图示总结四种变换的叠加规律和参数作用顺序。
              \item 学生活动：绘制思维导图整理变换规律。
          \end{itemize}
          \item 课堂练习
          \begin{itemize}
              \item 教师活动： 布置函数变换题组(如: 将 \(y = \sin x\) 变换为 \(y = 2\sin \left( {3x - \frac{\pi }{4}}\right) + 1\) )。
              \item 学生活动：通过 GeoGebra 验证变换过程，写出分步变换的数学表达式。
          \end{itemize}
        \end{enumerate}
      \item 实验演示
      
        \hspace*{2em}教师活动：
        \begin{enumerate}
            \item 相位平移变换演示，见图 \ref{fig:相位平移演示界面示意图}。
            \begin{figure}[htbp]
              \centering
              \includegraphics[width=0.6\textwidth]{相位平移演示界面示意图}
              \caption{相位平移演示界面示意图}
              \label{fig:相位平移演示界面示意图}
            \end{figure}
            
              \begin{enumerate}
                \item 创建滑动条控制 \(\varphi\) 值(范围 \(- 2\pi\) 到 \(2\pi\) )。
                \item 输入函数 \(y = \sin \left( {x + \varphi }\right)\) ，展示 \(\varphi\) 变化时曲线的水平移动。
                \item 对比 \(\varphi\) 取正负值时移动方向的差异。
              \end{enumerate}
            \item 周期变换演示，见图 \ref{fig:周期变换演示界面示意图}。
            \begin{figure}[htbp]
              \centering
              \includegraphics[width=0.6\textwidth]{周期变换演示界面示意图}
              \caption{周期变换演示界面示意图}
              \label{fig:周期变换演示界面示意图}
            \end{figure}
            
              \begin{enumerate}
                \item 创建 \(\omega\) 滑动条(范围 0.5 到 4)。
                \item 输入函数 \(y = \sin \left( {x + \frac{\pi }{4}}\right)\) ，展示图象。
                \item 输入函数 \(y = \sin \left( {\omega x + \frac{\pi }{4}}\right)\) ，展示 \(\omega\) 对曲线疏密程度的影响。
              \end{enumerate}
            \item 振幅变换演示，见图 \ref{fig:振幅变换演示界面示意图}。
            \begin{figure}[htbp]
              \centering
              \includegraphics[width=0.6\textwidth]{振幅变换演示界面示意图}
              \caption{振幅变换演示界面示意图}
              \label{fig:振幅变换演示界面示意图}
            \end{figure}
            
              \begin{enumerate}
                \item 创建 \(A\) 滑动条(范围 -3 到 3)。
                \item 输入函数 \(y = \sin \left( {2x + \frac{\pi }{4}}\right)\) ，展示图象。
                \item 输入函数 \(y = A\sin \left( {2x + \frac{\pi }{4}}\right)\) ，展示振幅变化对波峰高度的影响。
                \item 演示 \(A\) 为负值时图象的翻转现象。
              \end{enumerate}
            \item 综合变换演示，见图 \ref{fig:综合变换演示界面示意图}。
            \begin{figure}[htbp]
              \centering
              \includegraphics[width=0.6\textwidth]{综合变换演示界面示意图}
              \caption{综合变换演示界面示意图}
              \label{fig:综合变换演示界面示意图}
            \end{figure}
            
              \begin{enumerate}
                \item 同时设置 \(A\) 、 \(\omega\) 、 \(\varphi\) 三个滑动条。
                \item 输入函数 \(y = A\sin \left( {\omega x + \varphi }\right)\) 。
                \item 依次调整各参数展示复合变换效果。
                \item 演示参数调整顺序对最终图象的影响。
              \end{enumerate}
        \end{enumerate}

        \hspace*{2em}学生活动：
        \begin{enumerate}
            \item 观察单一参数变化时的图象特征变化。
            \item 尝试预测多参数复合变换的最终图象形态。
            \item 记录不同参数组合下的函数表达式与对应图象特征。
            \item 自主设计参数组合创造特定波形。
        \end{enumerate}
    \end{enumerate}
  \item[结论] \hspace*{2em} 通过 GeoGebra 动态演示，学生能直观理解三角函数各参数对图象的影响规律，建立 “代数参数” 与 “几何变换” 的双向对应关系。实验教学有效突破了相位平移方向、周期计算等传统教学难点，培养了学生的动态几何直观和数形结合能力。
  \item[分析]
  \hspace*{2em} 可设计以下拓展思考题:
  \begin{enumerate}
    \item 当变换顺序不同时(如先平移后伸缩、先伸缩后平移)，最终图象是否相同？ 如何用数学表达式解释?
    \item 对于余弦函数 \(y = A\cos \left( {\omega x + \varphi }\right)\) ，其参数作用规律与正弦函数有何异同?
    \item 如果要求将 \(y = \sin x\) 变换为某个特定波形，如何逆向求解所需的参数组合?
  \end{enumerate}
\end{description}


\section{讲义3——圆锥曲线的形成实验讲义}

\begin{description}
  \item[题目] \hspace*{2em} 圆锥曲线形成的实验讲义
  \item[目的] \mbox{} \vspace*{-1.8em}
    \begin{enumerate}
      \item 知识目标
        \begin{enumerate}
          \item 认识平面截割双圆锥面可以形成圆、椭圆、双曲线、抛物线。
          \item 理解平面截割双圆锥面形成圆、椭圆、双曲线、抛物线的几何条件。
        \end{enumerate}
      \item 能力目标
        \begin{enumerate}
          \item 培养三维空间想象能力与平面截取立体图形的分析能力。
          \item 提升利用动态几何软件观察几何变换的实践能力。
        \end{enumerate}
      \item 情感目标
        \begin{enumerate}
          \item 感受几何之美，体会二维曲线与三维空间的深刻联系。
          \item 培养从实验现象到数学本质的探究精神。
        \end{enumerate}
    \end{enumerate}
  \item[器材] \hspace*{2em} 电脑，希沃白板软件
  \item[过程] \mbox{} \vspace{-1.8em}
    \begin{enumerate}[label=(\zhnum*)]
      \item 主要流程（预演）
        \begin{enumerate}[1.]
          \item 引入新课
          \begin{itemize}
              \item 教师活动：可通过展示古罗马引水渠抛物线拱、行星椭圆轨道等实例，引出 “这些曲线如何从圆锥产生”的核心问题。
              \item 学生活动：观察经典建筑中的曲线元素，猜想平面截取圆锥的可能形态。
          \end{itemize}
          \item 实验演示
          \begin{itemize}
              \item 教师活动：使用希沃白板 3D 工具动态演示平面截割双圆锥过程。
              \item 学生活动：记录不同截面角度下的曲线类型。
          \end{itemize}
          \item 定理阐述
          \begin{itemize}
              \item 教师活动：解析各类圆锥曲线的几何特征。
              \item 学生活动：建立截面角度与曲线类型的对应关系。
          \end{itemize}
          \item 知识小结
          \begin{itemize}
              \item 教师活动：用动态对比图总结四类圆锥曲线的形成条件。
              \item 学生活动：完成圆锥曲线分类表格(截面角度、曲线类型)。
          \end{itemize}
          \item 课堂练习
          \begin{itemize}
              \item 教师活动：提供不同截面角度要求判断曲线类型。
              \item 学生活动：通过希沃白板调整平面验证猜想，解释几何原理。
          \end{itemize}
        \end{enumerate}
      \item 实验演示
      
        \hspace*{2em}教师活动：
        \begin{enumerate}
            \item 圆形截面演示，见图 \ref{fig:圆形截面演示}。
            \begin{figure}[htbp]
              \centering
              \includegraphics[width=0.7\textwidth]{圆形截面演示}
              \caption{圆形截面演示}
              \label{fig:圆形截面演示}

              \begin{enumerate}
                \item 点击“圆”按钮，展示截面交线为圆形。
                \item 动态旋转模型验证各方向对称性。
              \end{enumerate}
            \item 椭圆截面演示，见图 \ref{fig:椭圆截面演示}。
            \begin{figure}[htbp]
              \centering
              \includegraphics[width=0.7\textwidth]{椭圆截面演示}
              \caption{椭圆截面演示}
              \label{fig:椭圆截面演示}
            \end{figure}
            
              \begin{enumerate}
                \item 点击 “椭圆” 按钮，展示截面交线为椭圆。
                \item 观察截面运动与截线情况。
              \end{enumerate}
            \item 抛物线截面演示，见图 \ref{fig:抛物线截面演示}。
            \begin{figure}[htbp]
              \centering
              \includegraphics[width=0.7\textwidth]{抛物线截面演示}
              \caption{抛物线截面演示}
              \label{fig:抛物线截面演示}
            \end{figure}
            
              \begin{enumerate}
                \item 点击 “抛物线” 按钮，展示截面交线为抛物线。
                \item 观察截面运动与截线情况。
              \end{enumerate}
            \item 双曲线截面演示，见图 \ref{fig:双曲线截面演示}。
            \begin{figure}[htbp]
              \centering
              \includegraphics[width=0.7\textwidth]{双曲线截面演示}

            \item 记录不同截面角度对应的曲线类型。
            \item 尝试手动调整截面平面创造特定曲线。

  \item[结论] \hspace*{2em} 通过希沃白板中 “数画板” 的三维动态演示，学生直观理解圆锥曲线的空间形成机制，建立 “截面角度一曲线类型一几何特性” 的完整认知体系。实验教学有效突破了传统二维图示的局限，培养了学生的空间解析能力。
  \item[分析]
  \hspace*{2em} 可设计以下拓展探究:
  \begin{enumerate}
    \item 当截面平面通过圆锥顶点时，会形成什么特殊图形？
    \item 如何通过截面实验解释圆锥曲线的第二定义，即到定点与定直线距离之比为常数?
    \item 结合本节所学，尝试了解 Dandelin 双球。
  \end{enumerate}
\end{description}
