\chapter{导数在高中数学常规题中的应用}

本章主要介绍导数在高中数学题中的常见考点和解题方法。

\section{研究函数的部分性质}

利用导数可以求函数的极值并判断函数的单调区间.此外函数求最值要从极值点和区间端点的函数值找.

现在我们通过下面这道例题具体分析.

\begin{example}
  \[
    f(x)=\left(x^{2}+a x+b\right) e^{x}(x \in R)
  \]
  \begin{enumerate}[(1)]
    \item 若 $a=2, b=-2$, 求函数 ${f}({x})$ 的极值;
    \item 有 $x=1$ 为 $f(x)$ 一极值点。试求出 $a$ 关于 $b$ 的关系式（用 $a$ 表示 $b$ )。 讨论 ${f}(x)$ 的单调区间;
    \item 在（2）的条件下, 设 ${a}>0$, 函数 $g(x)=\left(a^{2}+14\right) e^{x+4}$, 若存在 $x_{1}, x_{2} \in[0,4]$ 使得 $\left|f\left(x_{1}\right)-g\left(x_{2}\right)\right| < 1$ 成立, 求 ${a}$ 的取值范围。
  \end{enumerate}
\end{example}

\begin{analysis}
  \begin{enumerate}[(1)]
    \item 属于比较基础的求导再求极值的问题, 先将条件给定的 $a$ 和 $b$ 值代人 函数 ${f}({x})$ 求出导函数 ${f}^{\prime}({x})$, 然后令 ${f}^{\prime}({x})=0$ 求实根, 接着判断实根是否是极值点, 最后求出极值；
    \item 由 $x=1$ 是函数 ${f}({x})$ 的一个极值点可知 ${f}^{\prime}(1)=0$ 接着列方程可求出 ${a}$ 关于 ${b}$ 的关系式; 求 ${f}({x})$ 的单调区间要先求出令 $f^{\prime}(x)=0$ 的根 1 和 $-3-a$ (这里需要注意因为 $x=1$ 是 $f(x)$ 的极 值点所以 $-3-a \neq 1)$ 即 $a \neq-4$, 接着讨论当 $a>-4$ 和 $a<-4$ 时 $f(x)$ 的单调性;
    \item 由(2)知: 当 $a>0$ 时, $f(x)$ 在 $(0,1)$ 上递减。在 $(1,4)$ 上递增。因此 ${f}({x})$ 的最小值在 ${x}=1$ 处取到, 而 ${f}({x})$ 的最大值则是判断 ${f}(0)$ 和 ${f}(4)$ 哪个值 更大。 又因为 $a>0$ 且 $y=e^{x}$ 是增函数, 所以 $g(x)$ 在 $[0,4]$ 上递增, 故最值在端点取到。若想存在 $x_{1}, x_{2} \in[0,4]$ 使得 $\left|f\left(x_{1}\right)-g\left(x_{2}\right)\right| < 1$ 成立, 只需 ${g}_{\text {min}}-f_{\text {max}}<1$ 成立即可。
  \end{enumerate}
\end{analysis}


\begin{proof}[解析]
  \begin{enumerate}[(1)]
    \item 
      \[f^{\prime}(x)=(2 x+a) e^{x}+\left(x^{2}+a x+b\right) e^{x}=\left[x^{2}+(2+a) x+(a+b)\right] e^{x}.\] 当 ${a}=2, {~b}=-2$ 时, $f(x)=\left(x^{2}+2 x-2\right) e^{x}$ 则 $f^{\prime}(x)=\left(x^{2}+4 x\right) e^{x}$.

      令 $f^{\prime}(x)=0$ 得 $\left(x^{2}+4 x\right) e^{x}=0$,  因为$e^{x} \neq 0$, 所以 $x^{2}+4 x=0$, 解得 $x_{1}=-4, x_{2}=0$.

      因为当 $x \in(-\infty,-4)$ 时, $f^{\prime}(x) > 0$,
      当 $x \in(-4,0)$ 时, $f^{\prime}(x) < 0$, 当 $x \in(0,+\infty)$ 时 $f^{\prime}(x) > 0$

      所以 当 $x=-4$ 时, 函数 $f(x)$ 有极大值, $f(x)_{\text {极大 }}=\frac{6}{e^{4}}$,
      当 $x=0$ 时, 函数 $f(x)$ 有极小值, $f(x)_{\text {极小 }}=-2$.
    \item 
      由(1)知 $f^{\prime}(x)=\left[x^{2}+(2+a) x+(a+b)\right] e^{x}$.

      因为$x=1$ 是函数 $f(x)$ 的一个极值点, 所以 $f^{\prime}(1)=0$,
      即e $[1+(2+a)+(a+b)]=0$, 解得 $b=-3-2 a$
      则 \[f^{\prime}(x)=e^{x}\left[x^{2}+(2+a) x+(-3-a)\right]=e^{x}(x-1)[x+(3+a)].\]
      令 $f^{\prime}(x)=0$, 得 $x_{1}=1$ 或 $x_{2}=-3-a$.

      因为$x=1$ 是极值点, 所以 $-3-a \neq 1$, 即 $a \neq-4$,
      当 $-3-a > 1$ 即 $a <-4$ 时, 由 $f^{\prime}(x) > 0$ 得 $x \in(-3-a,+\infty)$ 或 $x \in(-\infty, 1)$
      由 $f^{\prime}(x) < 0$ 得 $x \in(1,-3-a)$.

      当 $-3-a < 1$ 即 $a >-4$ 时, 由 $f^{\prime}(x) > 0$ 得 $x \in(1,+\infty)$ 或 $x \in(-\infty,-3-a)$
      由 $f^{\prime}(x) < 0$ 得 $x \in(-3-a, 1)$.
      
      综上所述:
      \begin{enumerate}[1)]
        \item 当 $a<-4$ 时, 递增区间是 $(-\infty, 1)$ 、 $(-3-a,+\infty)$。 递减区间是 $(1,-3-a)$;
        \item 当 $a>-4$ 时, 递增区间是 $(-\infty,-3-a)$ 、 $(1,+\infty)$。 递减区间是 $(-3-a, 1)$.
      \end{enumerate}

    \item 
      由(2)知: 当 $a>0$ 时, $f(x)$ 在 $(0,1)$ 上递减, 在 $(1,4)$ 上递增。
      所以$f(x)$ 在 $[0,4]$ 的最小值是 $f(1)=-(a+2) e$。

      又因为 $f(0)=b e^{x}=-(2 a+3)<0, f(4)=(2 a+13) e^{4}>0$,
      所以 函数 $f(x)$ 在 $[0,4]$ 的值域是 $\left[-(a+2) e,(2 a+13) e^{4}\right]$；$g(x)=\left(a^{2}+14\right) e^{x+4}$ 在 $[0,4]$ 是增函数，
      所以它在 $[0,4]$ 的值域为 $\left[\left(a^{2}+14\right) e^{4},\left(a^{2}+14\right) e^{8}\right]$.

      因为$\left(a^{2}+14\right) e^{4}-(2 a+13) e^{4}=\left(a^{2}-2 a+1\right) e^{4}=(a-1)^{2} e^{4} \geq 0$，
      所以 存在 $x_{1}, x_{2} \in[0,4]$ 使得 $\left|f\left(x_{1}\right)-g\left(x_{2}\right)\right|<1$ 成立只须
      \[
        \begin{aligned}
          \left(a^{2}+14\right) e^{4}-(2 a+13) e^{4}<1 &\Rightarrow(a-1)^{2} e^{4}<1\\
          &\Rightarrow(a-1)^{2}<\frac{1}{e^{4}} \Rightarrow 1-\frac{1}{e^{2}}<a<1+\frac{1}{e^{2}}. \qedhere
        \end{aligned}
      \]
  \end{enumerate}
\end{proof}

\begin{review}
  此类问题考查含参函数 ${f}({x})$ 单调性分析, 重难点在于对参数的合理分类及各 分类下导函数的符号研究。如果 ${f}^{\prime}({x})=0$ 解出来的实根中含有参数的话, 需要讨论 ${f}({x})$ 的定义域内是否包含该实根。
\end{review}



\section{利用导数的几何意义解题}

导数的几何意义可用于求函数在某处的切线方程.下面这道例题基本包括了如何利用它解题的方法.

\begin{example}
  \parencite{惠红民一题一课} 已知函数 $f(x)=x^{3}-4 x^{2}+5 x-4$.
  \begin{enumerate}[(1)]
    \item 求曲线 $f(x)$ 在 $x=1$ 处的切线方程;
    \item 求经过点 $A(2,-2)$ 的曲线 $f(x)$ 的切线方程;
    \item 若直线 $y=k x-4$ 与曲线 $f(x)$ 相切, 求 $k$ 的值.
  \end{enumerate}
\end{example}


\begin{analysis}
  \begin{enumerate}[(1)]
    \item 题目里 “求曲线在 $x=1$ 处的切线方程” 意味着所求直线与曲线 $f(x)$ 在 $x=1$ 处相切, 此时导数的几何意义派上用场.用此可得出该方程斜率. 具体步骤为求函数$f(x)$在$x=1$的导数. 再把 ${x}=1$ 代人 ${f}({x})$ 求出切点纵坐标, 用点斜式即可表示该切线方程;
    \item 需要注意点 ${A}(2,-2)$ 是否在曲线上, 是否为切点, 经验证 A $(2,-2)$ 在曲线 ${f}({x})$ 上, 设切点 $\left({x}_{0}, {x}_{0}^{3}-4 {x}_{0}^{2}+5 {x}_{0}-4\right)$, 再对 ${f}({x})$ 求导, 表示出斜率, 再用点斜式方程表示切线 (含 ${x}_{0}$ ), 而后利用切点既在切线上又在曲线上的特征列出方程求出切点坐标，然后就可以求出切线方程了;
    \item 已知直线 ${y}={kx}-4$与曲线 ${f}({x})$ 相切, 可先设切点 $\left({x}_{0}, {y}_{0}\right)$, 又因为直线 ${y}={kx}-4$ 的斜率 ${k}$ 可用曲线 ${f}
    (x)$ 的导函数 $f'(x)$ 来表示, 接着联立方程可求出切点进而求出斜率 ${k}$ 值。
  \end{enumerate}
\end{analysis}

\begin{proof}[解析]
  \begin{enumerate}[(1)]
    \item 因为 $f(x)=x^{3}-4 x^{2}+5 x-4$, 所以 $f^{\prime}(x)=3 x^{2}-8 x+5$, 所以 $f^{\prime}(1)=0$。
    又因为 $f(1)=-2$,
    所以曲线 $f(x)$ 在 ${x}=1$ 处的切线方程为 $y-(-2)=0 \times(x-1)$, 即 ${y}=-2$.
    \item 设切点坐标为 $\left(x_{0}, x_{0}^{3}-4 x_{0}^{2}+5 x_{0}-4\right), f^{\prime}(x)=3 x_{0}^{2}-8 x_{0}+5$, 则 切 线方程为 $y-(-2)=\left(3 x_{0}^{2}-8 x_{0}+5\right)(x-2)$。
    
    又因为切线过 $\left(x_{0}, x_{0}^{3}-4 x_{0}^{2}+5 x_{0}-4\right)$ 点,
    则 $x_{0}^{3}-4 x_{0}^{2}+5 x_{0}-2=\left(3 x_{0}^{2}-8 x_{0}+5\right)\left(x_{0}-2\right)$, 整理得 $\left(x_{0}-2\right)^{2}\left(x_{0}-1\right)=0$
    解得 $x_{0}=2$, 或 $x_{0}=1$。

    当 $x_{0}=2$ 时, 切线方程为 $x-y-4=0$;
    当 $x_{0}=1$ 时, 切线方程为 $y+2=0$。

    因此经过点 $A(2,-2)$ 的曲线 $f(x)$ 的切线方程为 $x-y-4=0$ 或 $y+2=0$.
    \item 设直线 $y=k x-4$ 与曲线 $f(x)$ 相切于点 $P\left(x_{0}, y_{0}\right)$,
    则
    \begin{align}
      {y}_{0}&={kx}_{0}-4  \label{quation:3.1}\\
      {y}_{0}&={x}_{0}^{3}-4 {x}_{0}^{2}+5 {x}_{0}-4,\label{quation:3.2}
    \end{align}
    又因为 $y^{\prime}=3 x^{2}-8 x+5$,
    所以 
    \begin{equation}\label{quation:3.3}
      k=\left.y^{\prime}\right|_{x=x 0}=3 x_{0}^{2}-8 x+5.
    \end{equation}
    由~\eqref{quation:3.1} \eqref{quation:3.2} \eqref{quation:3.3} 得 $\left(3 x_{0}^{2}-8 x_{0}+5\right) x_{0}-4=x_{0}^{3}-4 x_{0}^{2}+5 x_{0}-4$,
    所以 $3 {x}_{0}^{3}-8 {x}_{0}^{2}+5 {x}_{0}-4={x}_{0}^{3}-4 {x}_{0}^{2}+5 {x}_{0}-4$,
    即 $2 {x}_{0}^{3}-4 {x}_{0}^{2}=0$,
    $2 {x}_{0}^{2}\left({x}_{0}-2\right)=0$,
    故 $x_{0}=0$ 或 $x_{0}=2$,
    所以 $k=5$ 或 $k=1$. \qedhere
  \end{enumerate}
\end{proof}

\begin{review}
  本题考察了利用导数研究曲线$f(x)$上某点切线方程的知识点，另外解题时要分清题目所求的是“过某点的切线方程”还是“在曲线上某点处的切线方程”，要审清楚题中的条件是“在”点处还是“过”点。总之,无论是切线经过某一点还是已知切线斜率,切点坐标都是化解难题的关键所在。
\end{review}



\section{讨论不等式恒成立问题}

不等式恒成立问题是高考常考的题型，这类题往往很灵活，下面这道例题是其中一种类型. 要想解决这类题，学生还是需要多思多练.

\begin{example}
  \parencite{惠红民一题一课} 已知函数 ${f}({x})=x \ln x$
  \begin{enumerate}[(1)]
    \item 求 ${f}({x})$ 的单调区间;
    \item 当 $k \leq 1$ 时, 求证: $f(x) \geq k x-1$ 恒成立. 
  \end{enumerate}
\end{example}

\begin{analysis}
  \begin{enumerate}[(1)]
    \item 求 ${f}^{\prime}({x})$,列出 ${f}({x})$ 与 ${f}^{\prime}({x})$ 的区间表示即可得到的单调区间; 
    \item 对于第二 问这类不等式恒成立问题一般有两种方法, 一种是参数分离法, 以本题为例, 可将原 条件 “当 $k \leq 1$ 时, 求证: $f(x) \geq k x-1$ 恒成立.” 转化为 “ $k \leq 1$ 时, $\ln x+\frac{1}{x} \geq k$ 恒成立”， 故构造函数 $g(x)=\ln x+\frac{1}{x}$, 研究 ${g}({x})$ 的最值即可; 另一种方法是构造函数法, 本题可 构造函数 ${g}({x})={f}({x})-({kx}-1)$, 即研究 “ $k \leq 1$ 时, $g(x)=x \ln x-k x+1 \geq 0$ 恒成立”, 最后只要 证明 ${g}({x})$ 的最小值大于 0 即可。
  \end{enumerate}
\end{analysis}


\begin{proof}[解析]
  \begin{enumerate}[(1)]
    \item
      $f(x)$ 的定义域为 $(0,+\infty), f^{\prime}(x)=\ln x+1$, 令 $f^{\prime}(x)=0$, 得 $x=\frac{1}{e}$, $f^{\prime}(x)$ 与 $f(x)$ 随 $x$ 的变化情况如下表:

        \begin{tblr}{
          width = \textwidth,
          hlines, vlines,
          cells = {c,m},
          columns = {mode=dmath},
          colspec = {X[1]X[1]X[1]X[1]}
        }
          x & (0, \dfrac{1}{\upe}) & \dfrac{1}{\upe} & (\dfrac{1}{\upe}, +\infty) \\
          g'(x) & - & 0 & + \\
          g(x) & \searrow & \text{极小值} & \nearrow \\
        \end{tblr}

      所以 ${f}({x})$ 的单调递减区间为 $\left(0, \frac{1}{e}\right)$, 单调递增区间为 $\left(\frac{1}{e},+\infty\right)$ 。
    \item 方法一：
        设 $g(x)=\ln x+\frac{1}{x}, x>0, g^{\prime}(x)=\frac{1}{x}-\frac{1}{x^{2}}=\frac{x-1}{x^{2}}$.$g^{\prime}(x)$ 与 $g(x)$ 随 $x$ 的变化情况如下表:

        \begin{tblr}{
          width = \textwidth,
          hlines, vlines,
          cells = {c,m},
          columns = {mode=dmath},
          colspec = {X[1]X[1]X[1]X[1]}
        }
          x & (0, 1) & 1 & (1, +\infty) \\
          g'(x) & - & 0 & + \\
          g(x) & \searrow & \text{极小值} & \nearrow \\
        \end{tblr}

        所以 ${g}({x}) \geq {g}(1)=1$, 既在 $\ln x+\frac{1}{{x}} \geq 1$ 时恒成立,所以, 当 $k \leq 1$ 时, $\ln x+\frac{1}{x} \geq k$, 所以 $x \ln x+1 \geq k x$, 即 $x \ln x \geq k x-1$, 所以, 当 $k \leq 1$ 时, $f(x) \geq k x-1$ 恒成立.

      方法二：令 $g(x)=f(x)-(k x-1)=x \ln x-k x+1, g^{\prime}(x)=\ln x+1-k$, 令 $g^{\prime}(x)=0$, 得 $x= \upe ^ {k-1}$. $ g^{\prime}(x)$ 与 $g(x)$ 随 $x$ 的变化情况如下表:

      \begin{tblr}{
        width = \textwidth,
        hlines, vlines,
        cells = {c,m},
        columns = {mode=dmath},
        colspec = {X[1]X[1]X[1]X[1]}
      }
        x & (0, \upe ^ {k-1}) & \upe ^ {k-1} & (\upe ^ {k-1}, +\infty) \\
        g'(x) & - & 0 & + \\
        g(x) & \searrow & \text{极小值} & \nearrow \\
      \end{tblr}

      $g(x)$ 的最小值为 $g\left(\upe ^ {k-1}\right)=1-\upe ^ {k-1}$, 当 $k \leq 1$ 时, $\upe ^ {k-1} \leq 1$, 所以 $1-\upe ^ {k-1} \geq 0$, 故 $g(x) \geq 0$, 即当 $k \leq 1$ 时, $f(x) \geq k x-1$ 恒成立. \qedhere
  \end{enumerate}
\end{proof}



\section{讨论函数零点问题}

函数零点问题中的函数往往是含参函数，因此其函数较为复杂，需要先讨论该函数的单调性再判断零点个数.


\begin{example}
  现有函数 $f(x)=a \ln x+\frac{x^{2}}{2}-(a+1) x$。 当 $a \leq 1$ 时，探究 $f(x)$ 的零点个数。
\end{example}

\begin{analysis}
  本题在讨论函数单调性的基础上讨论函数的零点个数, 所以第一步是正确分 析函数单调性, 接着再比较极值与 0 的关系。当 $0<{a}<1$ 时较复杂, 需要画出函数图像 帮助分析零点情况。
\end{analysis}

\begin{proof}[解析]
  $f^{\prime}(x)=\frac{(x-1)(x-a)}{x}, x>0$,
  \begin{enumerate}[(1)]
    \item 
      当 $a \leq 0$ 时, $x \in(0,1)$ 时, $f^{\prime}(x)<0, f(x)$ 为减函数; $x \in(1,+\infty)$ 时, $f^{\prime}(x)>$ $0, {f}({x})$ 为增函数.
      所以 ${f}({x})$ 在 ${x}=1$ 时取到最小值 ${f}(1)=-{a}-\frac{1}{2}$.
      \begin{enumerate}[(i)]
        \item 当 $a=0$ 时,
        $f(x)=\frac{x^{2}}{2}-x$, 由于 $x>0$, 令 $f(x)=0 \Rightarrow x=2$, 则 $f(x)$ 在 $(0,+\infty)$ 上有一个零点;
        \item $a=-\frac{1}{2}$, 即 $f(1)=0$ 时, $f(x)$ 有一个零点;
        \item 当 $a<-\frac{1}{2}$, 即 $f(1)>0$ 时, 此时 $f(x)$ 无零点;
        \item 当 $-\frac{1}{2}<a<0$, 即 $f(1)<0$ 时, 由于 $x \rightarrow 0$ (从右侧趋近于 0 时)， $f(x) \rightarrow+\infty ; x \rightarrow+\infty$ 时 ,$f(x) \rightarrow+\infty$, 所以 $f(x)$ 有两个零点。
      \end{enumerate}
    \item 
      当 $0<a<1$ 时 , $x \in(0, a)$ 时, $f^{\prime}(x)>0, f(x)$ 为增函数; $x \in(a, 1)$ 时, $f^{\prime}(x)<$ $0, f(x)$ 为减函数; $x \in(1,+\infty)$ 时, $f^{\prime}(x)>0, f(x)$ 为增函数, 所以 $f(x)$ 在 $x=a$ 处取极大值, ${f}({x})$ 在 ${x}=1$ 处取极小值, 图像大致趋势如图~\ref{figure:3.4} 所示:
      
      \begin{figure}[htbp]
        \centering
        \includegraphics[width = 8cm]{3.png}
        \caption{}
        \label{figure:3.4}
      \end{figure}
    
      $f(a)=a \ln a+\frac{1}{2} a^{2}-(a+1) a=a \ln a-\frac{1}{2} a^{2}-a$。

      当 $0<{a}<1$ 时, ${f}({a})<0$, 即在 $x \in(0,1)$ 时, $f(x)<0$,
      而 $f(x)$ 在 $x \in(1,+\infty)$ 时为增函数, 且 $x \rightarrow+\infty$ 时, $f(x) \rightarrow+\infty$,
      所以此时 $f(x)$ 只有一个零点。
    \item 
      当 $a=1$ 时,
      $f^{\prime}(x)=\frac{(x-1)^{2}}{x} \geq 0$ 在 $(0,+\infty)$ 上恒成立, 所以 $f(x)$ 为增函数, 且 $x \rightarrow 0$ (从右侧趋近于 0) 时, $f(x) \rightarrow-\infty ; x \rightarrow+\infty$ 时, $f(x) \rightarrow+\infty$,
      所以 $f(x)$ 有一个零点。
  \end{enumerate}

  综上所述, 
  \begin{enumerate}[(1)]
    \item 当 $0 \leq a \leq 1$ 或 $a=-\frac{1}{2}$ 时, $f(x)$ 有一个零点; 
    \item 当 $a<\frac{1}{2}$ 时， $f(x)$ 无零点;
    \item 当 $-\frac{1}{2}<a<0$ 时, $f(x)$ 有两个零点。 \qedhere
  \end{enumerate}
\end{proof}



\section{讨论函数中的存在性问题}

对于函数中的存在性问题，通过导数研究具体例题，将问题转化为值域关系或者最值关系.

\begin{example}
  现有函数 $f(x)=-x^{3}+a x^{2}-4(a \in \mathbb{R})$。 若存在 $x_{0} \in(0,+\infty)$, 使 $f\left(x_{0}\right)>0$, 试找出 $a$ 的范围;
\end{example}

\begin{analysis}
  本题有两种方法, 一种是对参数分类讨论, 第二种是分离参数 $a$ 后求已知函 数最值。
\end{analysis}


\begin{proof}[解析]
  方法一: 存在 $x_{0} \in(0,+\infty)$, 使 $f\left(x_{0}\right)>0 \Leftrightarrow f(x)_{\max }>0, x \in(0,+\infty), {f}^{\prime}({x})=-3 {x}^{2}+2 {ax}$, 令 $f^{\prime}(x)=0$, 得 $x=0$ 或 $x=\frac{2}{3} a$,
  \begin{enumerate}[(1)]
    \item 当 ${a}<0$ 时, 因为 $f^{\prime}(x)<0, x \in(0,+\infty)$, 所以 $f(x)$ 在 $(0,+\infty)$ 上单调递增,
    所以 ${f}({x})<{f}(0)=-4$, 与 ${f}({x})_{\max }>0, {x} \in(0,+\infty)$ 矛盾, 所以 ${a}<0$ 不符合题意。
    \item 当 ${a}=0$ 时, 因为 ${f}^{\prime}({x})<0, {x} \in(0,+\infty)$, 所以 ${f}({x})$ 在 ${x} \in(0,+\infty)$ 上单调递减,
    所以 ${f}({x})<{f}(0)=-4$, 与 ${f}({x})_{\max}>0$ 矛盾, 所以 ${a}=0$ 不符合题意,
    \item 当 ${a}>0$ 时, 当 $x \in\left(0, \frac{2}{3} a\right)$ 时, $f^{\prime}(x)>0, x \in\left(\frac{2}{3} a,+\infty\right)$ 时, $f^{\prime}(x)<0$, 所以 $f(x)$ 在 $\left(0, \frac{2}{3} a\right)$ 上单调递增, 在 $\left(\frac{2}{3} a,+\infty\right)$ 上单调递增, 故令 ${f}({x})_{\max} >0$, 解得 ${a}>3$. 综上, $a>3$.
  \end{enumerate}

  方法二 (分离参数法) :
  因为存在 $x_{0} \in(0,+\infty)$, 使 $-x_{0}^{3}+a x_{0}^{2}-4>0$, 即存在 $x_{0} \in(0,+\infty)$, 使 $a>x_{0}+\frac{4}{x_{0}^{2}}$, 设 $g(x)=x+\frac{4}{x^{2}}, x \in(0,+\infty)$, 所以 $a>g(x)_{\min}$. \[g^{\prime}(x)=1-\frac{8}{x^{3}}=\frac{x^{3}-8}{x^{3}},\] 令 ${g}^{\prime}({x})=0$, 得 ${x}=2$. 
  
  当 $x \in(0,2)$ 时, ${g}^{\prime}({x})<0$, 当 $x \in(2,+\infty)$ 时 $g^{\prime}(x)>0$, 所以 $g(x)$ 在 $(0,2)$ 上单调递减, 在 $(2,+\infty)$ 上 单调递增, 所以 ${g}({x})={g}(2)=3$, 所以 ${a}>3$.
\end{proof}

\begin{review}
  能否进行参数分离取决于自变量的取值范围, 需要注意的是, 在问题转化时应 保证是等价转化. 存在性问题的各种不同情况要注意识别和区分。
\end{review}
